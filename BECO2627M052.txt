# Uber fare prediction: uses a CSV with pickup_longitude, pickup_latitude,
# dropoff_longitude, dropoff_latitude, passenger_count, fare_amount.
# If no CSV is supplied, a small synthetic dataset is generated for demonstration.
import os, numpy as np, pandas as pd
from sklearn.model_selection import train_test_split
from sklearn.ensemble import RandomForestRegressor
from sklearn.linear_model import LinearRegression
from sklearn.metrics import mean_squared_error, r2_score
from sklearn.impute import SimpleImputer
from sklearn.pipeline import make_pipeline
from sklearn.preprocessing import StandardScaler

path = "uber.csv"
if os.path.exists(path):
    df = pd.read_csv(path)
else:
    rng=np.random.default_rng(7); n=1000
    plon=rng.uniform(-74.05,-73.75,n); plat=rng.uniform(40.60,40.90,n)
    dlon=rng.uniform(-74.05,-73.75,n); dlat=rng.uniform(40.60,40.90,n)
    dist=np.sqrt((plon-dlon)**2+(plat-dlat)**2)*70
    df=pd.DataFrame({"pickup_longitude":plon,"pickup_latitude":plat,
      "dropoff_longitude":dlon,"dropoff_latitude":dlat,
      "passenger_count":rng.integers(1,6,n),
      "fare_amount":3+2.2*dist+rng.normal(0,2,n)})
features=["pickup_longitude","pickup_latitude","dropoff_longitude",
          "dropoff_latitude","passenger_count"]
df=df[features+["fare_amount"]].replace([np.inf,-np.inf],np.nan).dropna(subset=["fare_amount"])
X_train,X_test,y_train,y_test=train_test_split(df[features],df.fare_amount,test_size=.2,random_state=42)
models={"Linear Regression":make_pipeline(SimpleImputer(),StandardScaler(),LinearRegression()),
        "Random Forest":make_pipeline(SimpleImputer(),RandomForestRegressor(n_estimators=150,random_state=42))}
for name,model in models.items():
    model.fit(X_train,y_train); pred=model.predict(X_test)
    rmse=np.sqrt(mean_squared_error(y_test,pred))
    print(f"{name}: R2={r2_score(y_test,pred):.3f}, RMSE={rmse:.3f}")
print("Note: replace uber.csv with the actual cleaned dataset for real results.")

# Email classification: Normal (ham) vs Spam using Naive Bayes, KNN and SVM.
import pandas as pd
from sklearn.model_selection import train_test_split
from sklearn.feature_extraction.text import TfidfVectorizer
from sklearn.pipeline import make_pipeline
from sklearn.naive_bayes import MultinomialNB
from sklearn.neighbors import KNeighborsClassifier
from sklearn.svm import LinearSVC
from sklearn.metrics import classification_report, accuracy_score

# Optional CSV: columns must be "text" and "label" (ham/spam).
try:
    df=pd.read_csv("emails.csv")
    X=df["text"].astype(str); y=df["label"].astype(str).str.lower()
except FileNotFoundError:
    X=pd.Series([
      "Congratulations you won a free prize click now","Meeting moved to 3 pm tomorrow",
      "Claim your cash reward now","Please review the attached project report",
      "You have won lottery send bank details","Can we discuss the assignment today",
      "Urgent claim your free gift","The lecture starts at ten"
    ]*20)
    y=pd.Series(["spam","ham","spam","ham","spam","ham","spam","ham"]*20)
X_train,X_test,y_train,y_test=train_test_split(X,y,test_size=.25,random_state=42,stratify=y)
models={"Naive Bayes":make_pipeline(TfidfVectorizer(),MultinomialNB()),
        "K-Nearest Neighbors":make_pipeline(TfidfVectorizer(),KNeighborsClassifier(n_neighbors=3)),
        "Support Vector Machine":make_pipeline(TfidfVectorizer(),LinearSVC())}
for name,model in models.items():
    model.fit(X_train,y_train); pred=model.predict(X_test)
    print(f"\n{name} accuracy: {accuracy_score(y_test,pred):.3f}")
    print(classification_report(y_test,pred,zero_division=0))

# Bank customer churn prediction with an MLP neural-network classifier.
import os, numpy as np, pandas as pd
from sklearn.model_selection import train_test_split
from sklearn.compose import ColumnTransformer
from sklearn.preprocessing import OneHotEncoder, StandardScaler
from sklearn.pipeline import make_pipeline
from sklearn.impute import SimpleImputer
from sklearn.neural_network import MLPClassifier
from sklearn.metrics import accuracy_score, classification_report

if os.path.exists("bank.csv"):
    df=pd.read_csv("bank.csv")
    target="Exited" if "Exited" in df.columns else "churn"
    y=df[target]; X=df.drop(columns=[target])
else:
    rng=np.random.default_rng(3); n=800
    age=rng.integers(18,80,n); balance=rng.uniform(0,200000,n)
    tenure=rng.integers(0,11,n); products=rng.integers(1,5,n)
    geography=rng.choice(["France","Spain","Germany"],n)
    gender=rng.choice(["Female","Male"],n)
    logit=-3+0.035*(age-40)+0.000012*balance+0.45*(products==1)+0.7*(geography=="Germany")
    prob=1/(1+np.exp(-logit)); y=(rng.random(n)<prob).astype(int)
    X=pd.DataFrame({"Age":age,"Balance":balance,"Tenure":tenure,"NumOfProducts":products,
                    "Geography":geography,"Gender":gender})
numeric=X.select_dtypes(include=np.number).columns.tolist()
categorical=[c for c in X.columns if c not in numeric]
prep=ColumnTransformer([
 ("num",make_pipeline(SimpleImputer(strategy="median"),StandardScaler()),numeric),
 ("cat",make_pipeline(SimpleImputer(strategy="most_frequent"),OneHotEncoder(handle_unknown="ignore")),categorical)])
model=make_pipeline(prep,MLPClassifier(hidden_layer_sizes=(32,16),max_iter=600,random_state=42,early_stopping=True))
Xtr,Xte,ytr,yte=train_test_split(X,y,test_size=.2,random_state=42,stratify=y)
model.fit(Xtr,ytr); pred=model.predict(Xte)
print("Accuracy:",round(accuracy_score(yte,pred),3))
print(classification_report(yte,pred,zero_division=0))

# Gradient Descent for f(x) = x^2 + 4x + 5; minimum is at x = -2.
def f(x): return x*x + 4*x + 5
def derivative(x): return 2*x + 4
x=float(input("Initial x: "))
learning_rate=float(input("Learning rate (e.g. 0.1): "))
iterations=int(input("Number of iterations: "))
for i in range(iterations):
    x -= learning_rate*derivative(x)
    if i % max(1,iterations//10)==0:
        print(f"Iteration {i+1}: x={x:.6f}, f(x)={f(x):.6f}")
print(f"Approximate minimum: x={x:.6f}, f(x)={f(x):.6f}")

# K-Means and Agglomerative Hierarchical clustering on sales_data_sample.csv.
# If CSV is absent, a sample dataset is generated.
import os, numpy as np, pandas as pd
import matplotlib.pyplot as plt
from sklearn.preprocessing import StandardScaler
from sklearn.cluster import KMeans, AgglomerativeClustering
from sklearn.metrics import silhouette_score
from sklearn.decomposition import PCA
if os.path.exists("sales_data_sample.csv"):
    df=pd.read_csv("sales_data_sample.csv")
    numeric=df.select_dtypes(include=np.number).dropna(axis=1,how="all").fillna(0)
else:
    rng=np.random.default_rng(12)
    numeric=pd.DataFrame(np.vstack([rng.normal([20,30],[4,5],(80,2)),
                                    rng.normal([60,70],[5,4],(80,2)),
                                    rng.normal([90,20],[4,5],(80,2))]),
                         columns=["Annual_Spend","Orders"])
if numeric.shape[1]<2: raise ValueError("Dataset must contain at least two numeric columns.")
X=StandardScaler().fit_transform(numeric)
scores=[]
for k in range(2,8):
    labels=KMeans(n_clusters=k,random_state=42,n_init=10).fit_predict(X)
    scores.append((k,silhouette_score(X,labels)))
best_k=max(scores,key=lambda z:z[1])[0]
print("Silhouette scores:",scores,"\nSelected number of clusters:",best_k)
km=KMeans(n_clusters=best_k,random_state=42,n_init=10).fit_predict(X)
hc=AgglomerativeClustering(n_clusters=best_k).fit_predict(X)
coords=PCA(n_components=2).fit_transform(X)
fig,axes=plt.subplots(1,2,figsize=(10,4))
axes[0].scatter(coords[:,0],coords[:,1],c=km);axes[0].set_title("K-Means")
axes[1].scatter(coords[:,0],coords[:,1],c=hc);axes[1].set_title("Hierarchical")
plt.tight_layout();plt.savefig("clustering_result.png",dpi=150);plt.show()
